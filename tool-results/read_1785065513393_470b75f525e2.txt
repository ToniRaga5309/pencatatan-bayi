   544→function RecentRecords() {
   545→  const [records, setRecords] = useState<Array<{
   546→    id: string
   547→    namaBayi: string
   548→    namaIbu: string
   549→    tanggalLahir: string

   551→    status: string

   557→    fetchRecentRecords()
   558→  }, [])

   560→  const fetchRecentRecords = async () => {
   561→    try {
   562→      const response = await fetch("/api/operator/recent-records")
   563→      if (response.ok) {
   564→        const data = await response.json()
   565→        setRecords(data)
   566→      }
   567→    } catch (error) {
   568→      console.error("Error fetching recent records:", error)
   569→    } finally {
   570→      setIsLoading(false)
   571→    }
   572→  }
   573→
   574→  const getStatusBadge = (status: string) => {
   575→    switch (status) {
   576→      case "VERIFIED":
   577→        return (
   578→          <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 text-xs">
   579→            <CheckCircle className="w-3 h-3 mr-1" />Terverifikasi
   580→          </Badge>
   581→        )
   582→      case "PENDING":
   583→        return (

   585→            <Clock className="w-3 h-3 mr-1" />Menunggu
   586→          </Badge>
   587→        )
   588→      case "REJECTED":

   590→          <Badge variant="destructive" className="text-xs">

   594→      default:

   597→  }
   598→

   601→      <div className="flex justify-center py-8">

   621→        <div

   623→          className="flex items-center justify-between p-3 bg-slate-50 rounded-lg hover:bg-slate-100 transition-colors"

   630→                NIK Bayi: {record.nikBayi}
